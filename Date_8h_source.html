288 void addMonths(int n)
289 {
290 int y = 0;
291 unsigned m = 0, d = 0;
292 get(y, m, d);
293
294 y += n / 12;
295
296 unsigned deltaM = static_cast<unsigned>( std::abs(n) % 12 );
297
298 if(n < 0)
299 {
300 if(m <= deltaM)
301 {
302 m += 12;
303 y -= 1;
304 }
305
306 m -= deltaM;
307 }
308 else
309 {
310 m += deltaM;
311
312 if(m > 12)
313 {
314 m -= 12;
315 y += 1;
316 }
317 }
318
319 Date dt(y, m, 1);
320 unsigned maxDays = dt.daysInMonth();
321 if(d > maxDays)
322 {
323 d = maxDays;
324 }
325
326 set(y, m, d);
327 }

517inline unsigned Date::daysInMonth() const
518{

520 {
521 0,31,28,31,30,31,30,31,31,30,31,30,31
522 };
523
