133template <typename R, typename Alloc = Allocator>
134class BasicService : public Service
135{
136 public:
139 BasicService()
140 { }
141
144 ~BasicService()
145 { }
146
147 protected:
148 virtual Responder* onGetResponder(const Request&)
149 {
150 void* r = _alloc.allocate( sizeof(R) );
151 return new(r) R(*this);
152 }
153
154 virtual void onReleaseResponder(Responder* r)
155 {
156 r->~Responder();
157 _alloc.deallocate( r, sizeof(R) );
158 }
159
160 private:
161 Alloc _alloc;
162};
