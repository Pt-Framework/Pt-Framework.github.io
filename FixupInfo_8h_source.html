135template <typename T>
136inline void fixup(const FixupInfo& fixup, T*& fixme)
137{
138 fixme = 0;
139
140 if( ! fixup.isNull() )
141 {
142 fixme = fixup.getTarget<T>();
143 }
144}

151template <typename T>
152inline void fixup(const FixupInfo& fixup, T& fixme)
153{
154 T* to = fixup.getTarget<T>();
155 fixme = *to;
156}
