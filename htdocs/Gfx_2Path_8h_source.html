322  const Path::Element& operator*() const

330  {
331  return &_element;
332  }
333 
336  PathIterator& operator++()
337  {
338  if( _entry->size() > 0)
339  {
340  const PointF& pos = _points[ _entry->size() - 1 ];
341  _element.setPosition(pos);
342  }
343 
344  _points += _entry->size();
345  ++_entry;
346 
347  _element.setEntry(_entry, _points);
348  return *this;
349  }
350 
353  PathIterator operator++(int)
354  {
355  PathIterator tmp = *this;
356  ++(*this);
357  return tmp;
358  }
