256 template <typename CharT, typename TraitsT>
257 inline std::streamsize BasicOStream<CharT, TraitsT>::writesome(CharT* buffer, std::streamsize n)
258 {
259  if( ! _buffer || this->rdbuf() != _buffer )
260  return 0;
261 
262  std::streamsize avail = _buffer->out_avail();
263  if(avail == 0)
264  {
265  return 0;
266  }
267 
268  n = std::min(avail, n);
269  return _buffer->sputn(buffer, n);
270 }
