137 {

139 }
140
141 private:
142 template <typename T>
143 using ValueType = typename TypeTraits<T>::Value;
144
145 BasicDecomposer< ValueType<R> > _r;
146 Arguments<ValueType<As>...> _argv;
147 Composer* _args[sizeof...(As) + 1];
